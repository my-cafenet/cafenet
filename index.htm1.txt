<script>

function toggleMenu() {

    document
        .getElementById("nav")
        .classList
        .toggle("open");

}


document
    .querySelectorAll("nav a")
    .forEach(link => {

        link.addEventListener("click", () => {

            document
                .getElementById("nav")
                .classList
                .remove("open");

        });

    });


const observer =
    new IntersectionObserver(

        entries => {

            entries.forEach(entry => {

                if (entry.isIntersecting) {

                    entry.target
                        .classList
                        .add("show");

                    observer
                        .unobserve(entry.target);

                }

            });

        },

        {
            threshold: .12
        }

    );


document
    .querySelectorAll(".reveal")
    .forEach(el => {

        observer.observe(el);

    });
